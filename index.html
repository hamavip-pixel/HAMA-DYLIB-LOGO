async function scan(buf) { const d = new Uint8Array(buf); const found = []; // Known image location const knownOffset = 6681009; const knownBytes = 9284; if (knownOffset + knownBytes <= d.length) { const knownBlob = new Blob( [d.slice(knownOffset, knownOffset + knownBytes)], { type: 'image/jpeg' } ); const knownImage = await validImage(knownBlob); if (knownImage) { found.push({ start: knownOffset, end: knownOffset + knownBytes, blob: knownBlob, fmt: 'JPEG', w: knownImage.w, h: knownImage.h }); knownImage.bitmap.close(); } } // Scan other JPEG images for (let i = 0; i < d.length - 2;) { if (d[i] === 255 && d[i+1] === 216 && d[i+2] === 255) { let e = i + 3; while ( e < d.length - 1 && !(d[e] === 255 && d[e+1] === 217) ) e++; if (e < d.length - 1) { const end = e + 2; const blob = new Blob( [d.slice(i, end)], { type: 'image/jpeg' } ); const v = await validImage(blob); if (v) { found.push({ start: i, end, blob, fmt: 'JPEG', w: v.w, h: v.h }); v.bitmap.close(); } i = end; } else { i += 3; } } else { i++; } } found.sort((a, b) => a.start - b.start); return found.filter((x, i) => !found.slice(0, i).some( y => x.start < y.end && x.end > y.start ) ); }
